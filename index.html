#!/bin/zsh
# -------------------------------------------------------------------
# Script: Jamf Pro Automated Policy Execution & Health Check
# Engineer: TheScriptingTree (Sydney, AU)
# -------------------------------------------------------------------

LOG_FILE="/var/log/thescriptingtree_deployment.log"

log_message() {
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] $1" | tee -a $LOG_FILE
}

# Verify root privileges
if [ $EUID -ne 0 ]; then
    log_message "ERROR: Script must be executed as root."
    exit 1
fi

log_message "Initiating Jamf binary policy check..."
/usr/local/jamf/bin/jamf policy -event "SydneyEDU_Standard_Config"
